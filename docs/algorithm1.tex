\noindent\textbf{Algorithm 1} \; Safety layer on top of a frozen flow policy (one action chunk)\par
\vspace{2pt}\hrule\vspace{4pt}
\noindent\textbf{Input:} observation $o_t$; frozen flow policy $\pi$ with velocity field $v_\theta$;
runtime safety constraints, given as a cost function $J(\hat{a};\,\mathrm{ctx})$ on the predicted action chunk,
with $\mathrm{ctx}$ the constraint context of the coming chunk; weight $\lambda$; $N$ Euler steps; chunk length $H$; executed steps $K$\par
\noindent\textbf{Output:} safe action chunk $a_{1:K}$\par
\vspace{2pt}
\begin{algorithmic}[1]
\State $\mathrm{cond} \gets \pi.\mathrm{prepare}(o_t)$ \Comment{\NOTE{vision-language conditioning, once per chunk}}
\State \SL{$\mathrm{ctx} \gets \mathrm{Supervisor.before\_chunk}()$} \Comment{\SL{get the robot and the obstacles context}}
\State $x \gets \varepsilon \sim \mathcal N(0, I)$ \Comment{\NOTE{noise, the policy's own sampler}}
\For{$i = 0, \ldots, N-1$} \Comment{\NOTE{the policy's Euler integration}}
    \State $t \gets t_i$; \; $v \gets v_\theta(\mathrm{cond}, x, t)$
    \State $\hat{a} \gets x - t \cdot v$ \Comment{\NOTE{predicted clean chunk (rectified flow)}}
    \State \SL{$J \gets J(\hat{a};\,\mathrm{ctx})$} \Comment{\SL{cost of the predicted chunk}}
    \If{\SL{$J > 0$}}
        \State \SL{$\hat{g} \gets \mathrm{normalise}\!\left(\partial J / \partial \hat{a}\right)$}
        \State \SL{$v \gets v + \lambda \cdot \hat{g}$} \CAR{$+\; \mathrm{gate}(\mathrm{ctx}) \cdot \mathrm{CAR}(u)$} \Comment{\SL{steer the flow;} \CAR{off-manifold correction CAR($u$)}}
    \EndIf
    \State $x \gets x + \Delta t \cdot v$
\EndFor
\State $a_{1:H} \gets \mathrm{unnormalize}(x)$; \; execute $a_{1:K}$; \; \SL{$\mathrm{Supervisor.after\_chunk}()$}
\end{algorithmic}
\vspace{2pt}\hrule
